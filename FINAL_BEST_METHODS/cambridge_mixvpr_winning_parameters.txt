Command template
----------------

PY=/home/andreas/anaconda3/envs/sam3/bin/python

while IFS='|' read -r TAG CFG ROOT SPLIT QRET; do
  BASE=outputs/${TAG}

  $PY -m plm_match.pipelines.lifted_nn_localize \
    --config "${BASE}/aliked_native_netvlad_sfm/config_aliked_native.yaml" \
    --dataset_root "$ROOT" \
    --split_json "$SPLIT" \
    --attached_index "${BASE}/aliked_native_netvlad_index_aligned" \
    --retrieval_file "$QRET" \
    --out_dir "${BASE}/aliked_plm_best_hloc_nn_pnp12_mixvpr10" \
    --method aliked_h5 \
    --db_features_path "${BASE}/aliked_features/db.h5" \
    --query_features_path "${BASE}/aliked_features/query.h5" \
    --landmark_match_mode image_obs_hloc_nn \
    --memory_score_weight 0.0 \
    --topk 10 \
    --query_topk 4096 \
    --pnp_first_thresh 12.0 \
    --pnp_refine_thresh 12.0 \
    --oracle_candidate_diagnostic \
    --support_weight 0.0 \
    --point_support_weight 0.0 \
    --rank_weight 0.0 \
    --attach_dist_weight 0.0
done <<'EOF'

Notes
-----

The command above uses NN/HLocNN inside PLM before PnP.
It does not run SuperGlue or LightGlue at query time.
MixVPR is used only as the image retrieval frontend.
